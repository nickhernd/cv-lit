\section{Instalación del entorno}

\begin{lstlisting}[language=bash]
conda create -n cv-lit python=3.11
conda activate cv-lit
pip install torch torchvision          # SAM
pip install segment-anything           # SAM (Meta)
pip install opencv-python numpy        # vision + algebra
pip install gdal pyproj shapely        # geo
pip install requests                   # API Obscape
\end{lstlisting}
