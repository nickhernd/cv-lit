% Chapter 5 — Módulo de calibración offline
\chapter{Módulo de Calibración Offline [3.0]}
\label{chap:calibration}

\section{Descripción}

Ejecución única por cámara (o cuando se reposicione). Es previa al pipeline
de producción y no requiere conectividad en tiempo real.

\section{Entradas}

\begin{itemize}
  \item \textbf{GCPs GNSS} del IEL: puntos de control con coordenadas
        $(X, Y, Z)$ en EPSG:25830 y su píxel correspondiente $(u, v)$.
        Mínimo 4 pares (recomendado $\geq 8$).
  \item \textbf{Imagen de referencia}: fotografía de la escena en
        condiciones estables.
\end{itemize}

\section{Proceso}

\begin{enumerate}
  \item \textbf{Corrección de distorsión}: estimación de parámetros
        intrínsecos con \texttt{cv2.calibrateCamera()}.
  \item \textbf{Emparejamiento} manual o semi-automático de GCPs con
        píxeles en imagen.
  \item \textbf{Estimación de homografía} con RANSAC:
        \[
          H_i = \texttt{cv2.findHomography}(
                \mathbf{pts\_img},\;\mathbf{pts\_world},\;
                \texttt{cv2.RANSAC},\;
                \texttt{ransacReprojThreshold}=5.0
                )
        \]
  \item \textbf{Validación}: error de reproyección sobre GCPs \textbf{no
        usados} en estimación. Umbral de calidad: RMSE $< 1.5$ px.
  \item \textbf{Serialización}: $H_i$ en \texttt{calibration/cam\_\{i\}\_H.npy}
        y parámetros en \texttt{calibration/cam\_\{i\}\_params.json}.
\end{enumerate}

\section{Dependencias}

\section{Salida}

Por cada cámara $i = 1 \ldots 6$:

\begin{itemize}
  \item \texttt{calibration/cam\_\{i\}\_H.npy} --- matriz $H_i \in
        \mathbb{R}^{3\times3}$.
  \item \texttt{calibration/cam\_\{i\}\_params.json} --- focal, centro
        óptico, coeficientes de distorsión, RMSE y fecha de calibración.
\end{itemize}
